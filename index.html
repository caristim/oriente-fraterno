async function handleNotify() { if (!('Notification' in window)) { showToast('Navegador no soportado'); return; } if (!window.isSecureContext) { showToast('Se requiere HTTPS'); return; } try { const registration = await navigator.serviceWorker.ready; if (!registration) { showToast('Service Worker no listo'); return; } if (Notification.permission === 'granted') { showToast('Notificaciones ya activas ✦'); return; } if (Notification.permission === 'denied') { showToast('Permiso bloqueado'); return; } const perm = await Notification.requestPermission(); setNotifPerm(perm); if (perm === 'granted') { events.forEach(scheduleNotification); await syncEventsToSW(events); showToast('Notificaciones activadas ✦'); } else { showToast('Permiso denegado'); } } catch (err) { console.error(err); showToast('Error al activar notificaciones'); } }
